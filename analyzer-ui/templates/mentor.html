parse_score raises ValueError on a bad field. Printing stays in the caller, not in this function.
def parse_score(line):
    """Return an int score from a 'name,score' row."""
    name, raw = line.split(",", 1)
    raw = raw.strip()
    if not raw.isdigit():
        raise ValueError(f"score for {name!r} is not a number: {raw!r}")
    return int(raw)
