Explanation
parse_score turns one comma-separated row into an integer score and raises ValueError when the score field is not a number. Printing stays with the caller.
def parse_score(line):
    """Return an int score from a 'name,score' row."""
    name, raw = line.split(",", 1)
    raw = raw.strip()
    if not raw.isdigit():
        raise ValueError(f"score for {name!r} is not a number: {raw!r}")
    return int(raw)
